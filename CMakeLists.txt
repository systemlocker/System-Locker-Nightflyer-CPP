cmake_minimum_required(VERSION 3.20)
project(SystemLockerNightflyer VERSION 0.1.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(SYSLOCKER_NIGHTFLYER_STATIC "Build a static library" ON)
option(SYSLOCKER_NIGHTFLYER_BUILD_EXAMPLE "Build the quickstart example" ON)
if(MSVC AND SYSLOCKER_NIGHTFLYER_STATIC AND VCPKG_TARGET_TRIPLET MATCHES "-static$")
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(CURL REQUIRED)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(nlohmann_json 3.11 REQUIRED)

if(SYSLOCKER_NIGHTFLYER_STATIC)
    set(SYSLOCKER_NIGHTFLYER_TYPE STATIC)
else()
    set(SYSLOCKER_NIGHTFLYER_TYPE SHARED)
endif()
add_library(systemlocker_nightflyer ${SYSLOCKER_NIGHTFLYER_TYPE} src/nightflyer.cpp src/curl_transport.cpp src/authorization_session.cpp src/dpapi_state_store.cpp src/windows_cng_key_provider.cpp)
add_library(SystemLocker::Nightflyer ALIAS systemlocker_nightflyer)
set_target_properties(systemlocker_nightflyer PROPERTIES EXPORT_NAME Nightflyer)
target_include_directories(systemlocker_nightflyer PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(systemlocker_nightflyer PUBLIC CURL::libcurl OpenSSL::Crypto nlohmann_json::nlohmann_json)
if(WIN32)
    target_link_libraries(systemlocker_nightflyer PRIVATE Crypt32 Ncrypt Bcrypt Ole32)
endif()
target_compile_options(systemlocker_nightflyer PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/W4 /permissive-> $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall -Wextra -Wpedantic>)
if(SYSLOCKER_NIGHTFLYER_BUILD_EXAMPLE)
    add_executable(systemlocker_nightflyer_quickstart examples/authorize.cpp)
    target_link_libraries(systemlocker_nightflyer_quickstart PRIVATE SystemLocker::Nightflyer)
endif()

install(TARGETS systemlocker_nightflyer EXPORT SystemLockerNightflyerTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT SystemLockerNightflyerTargets FILE SystemLockerNightflyerTargets.cmake NAMESPACE SystemLocker:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SystemLockerNightflyer)
configure_package_config_file(cmake/SystemLockerNightflyerConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/SystemLockerNightflyerConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SystemLockerNightflyer)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SystemLockerNightflyerConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SystemLockerNightflyerConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/SystemLockerNightflyerConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SystemLockerNightflyer)
